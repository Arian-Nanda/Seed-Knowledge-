#!/usr/bin/env python3
"""
wiki_fetch_mobs.py

Fetches raw wikitext for every mob page from minecraft.wiki's public
MediaWiki API, and saves each one as a plain text file. This is Phase 1
of pulling wiki content into our knowledge base - starting with mob
behavior, since that's content we genuinely don't have yet (our existing
facts cover blocks/items/recipes/mechanics, not mob AI/combat behavior).

WHY WIKITEXT, NOT RENDERED HTML: wikitext preserves the page's actual
structure (== Headings ==, {{Infobox}} templates) which makes it much
easier to reliably pull out specific sections later (like "Behavior" or
"Drops") without fighting with HTML layout/styling noise.

RESPECTFUL USAGE: this script identifies itself with a real User-Agent
(required by the Wikimedia API etiquette most MediaWikis follow) and
waits between requests rather than hammering the server. If this is your
first time running it and something looks wrong, stop and let's fix it
together rather than re-running repeatedly.

LICENSE NOTE: minecraft.wiki content is CC BY-NC-SA. We're pulling raw
text here for OUR OWN processing into short factual statements (not
copying prose verbatim into the final knowledge base) - similar to how
we've handled every other data source. Attribution will be added to the
final facts, matching how we credited Faithful for textures.

USAGE:
    pip install requests --break-system-packages
    python3 wiki_fetch_mobs.py

OUTPUT:
    Creates a "wiki_raw/mobs/" folder with one .txt file per mob page.
"""

import requests
import time
import os
import json

API_URL = "https://minecraft.wiki/api.php"
OUTPUT_DIR = "wiki_raw/mobs"

# Identify ourselves honestly - required by API etiquette, and just good
# practice for automated access to someone else's server.
HEADERS = {
    "User-Agent": "SmartSeed-Personal-Project/1.0 (personal Minecraft fan project, non-commercial; contact via GitHub Arian-Nanda)"
}


def get_category_members(category, cmtype="page", limit=500):
    """Gets every page title in a given wiki category, handling pagination."""
    members = []
    cmcontinue = None
    while True:
        params = {
            "action": "query",
            "list": "categorymembers",
            "cmtitle": f"Category:{category}",
            "cmlimit": limit,
            "cmtype": cmtype,
            "format": "json",
        }
        if cmcontinue:
            params["cmcontinue"] = cmcontinue

        resp = requests.get(API_URL, params=params, headers=HEADERS, timeout=15)
        resp.raise_for_status()
        data = resp.json()

        batch = data.get("query", {}).get("categorymembers", [])
        members.extend(p["title"] for p in batch)

        if "continue" in data:
            cmcontinue = data["continue"]["cmcontinue"]
            time.sleep(0.5)  # be polite between paginated requests
        else:
            break

    return members


def get_page_wikitext(title):
    """Fetches the raw wikitext (markup) for a single page title."""
    params = {
        "action": "query",
        "titles": title,
        "prop": "revisions",
        "rvprop": "content",
        "rvslots": "main",
        "format": "json",
    }
    resp = requests.get(API_URL, params=params, headers=HEADERS, timeout=15)
    resp.raise_for_status()
    data = resp.json()

    pages = data.get("query", {}).get("pages", {})
    for page_id, page in pages.items():
        if page_id == "-1":
            return None  # page doesn't exist
        revisions = page.get("revisions", [])
        if revisions:
            return revisions[0]["slots"]["main"]["*"]
    return None


def safe_filename(title):
    """Converts a wiki page title into a safe filename."""
    return title.replace("/", "_").replace(" ", "_").replace(":", "_") + ".txt"


def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    # "Category:Mobs" turned out to only contain overview/classification
    # pages (Boss, Animal, Undead, etc.), not individual creatures - a
    # common wiki-categorization pattern where individual mobs are instead
    # split across categories like these. Checking several candidates at
    # once so we get a clear picture in a single run.
    CANDIDATE_CATEGORIES = [
        "Hostile mobs",
        "Passive mobs",
        "Neutral mobs",
        "Renewable mobs",
        "Unrenewable mobs",
        "Utility mobs",
        "List of mobs",
    ]

    all_titles = set()
    print("Checking candidate categories for individual mob pages...")
    for cat in CANDIDATE_CATEGORIES:
        try:
            titles = get_category_members(cat)
        except Exception as e:
            print(f"  Category:{cat} -> ERROR: {e}")
            continue
        print(f"  Category:{cat} -> {len(titles)} pages found. Sample: {titles[:5]}")
        all_titles.update(titles)
        time.sleep(0.3)

    mob_titles = sorted(all_titles)
    print()
    print(f"Combined total (deduplicated): {len(mob_titles)} pages.")

    if len(mob_titles) == 0:
        print("Still found nothing - none of these category names matched.")
        print("Please check https://minecraft.wiki/w/Zombie directly in a")
        print("browser, scroll to the bottom to see what categories THAT")
        print("page actually belongs to, and let's try those exact names.")
        return

    print()
    print("Fetching each page's content (this will take a few minutes)...")

    fetched = 0
    skipped = 0
    for i, title in enumerate(mob_titles):
        out_path = os.path.join(OUTPUT_DIR, safe_filename(title))
        if os.path.exists(out_path):
            skipped += 1
            continue  # already fetched in a previous run

        try:
            content = get_page_wikitext(title)
        except Exception as e:
            print(f"  [{i+1}/{len(mob_titles)}] ERROR on '{title}': {e}")
            continue

        if content is None:
            print(f"  [{i+1}/{len(mob_titles)}] SKIPPED (no content): {title}")
            continue

        with open(out_path, "w", encoding="utf-8") as f:
            f.write(content)
        fetched += 1
        print(f"  [{i+1}/{len(mob_titles)}] Saved: {title} ({len(content)} chars)")

        time.sleep(0.5)  # be polite between page requests

    print()
    print(f"Done. Fetched {fetched} new pages, skipped {skipped} already-fetched pages.")
    print(f"All content saved in: {OUTPUT_DIR}/")


if __name__ == "__main__":
    main()
